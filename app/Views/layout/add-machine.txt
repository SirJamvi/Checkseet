<script src="assets/js/jquery.min.js"></script>

<script>
$(document).ready(function() {
    
    if($('#msgsuccess').html() != "Kosong"){
        alert($('#msgsuccess').html());
    }

    // Fungsi terpusat untuk memuat proses
    function loadProcesses() {
        var deviceCode = $('#device').val();
        var docType = $('#type').val();
        var processSelect = $('#process_code');

        processSelect.empty().append('<option value="">Memuat data...</option>');
        processSelect.prop('disabled', true);

        // AJAX hanya berjalan jika Device DAN Tipe sudah dipilih
        if (deviceCode !== '' && docType !== '') {
            $.ajax({
                url: "<?= base_url('machine/get-processes') ?>",
                type: "GET",
                data: { device_code: deviceCode, type: docType }, // Kirim 2 parameter
                dataType: "JSON",
                success: function(response) {
                    processSelect.empty();
                    if(response.length > 0) {
                        processSelect.append('<option value="" selected>-- Pilih Process --</option>');
                        $.each(response, function(index, item) {
                            processSelect.append('<option value="' + item.process_code + '">' + item.name + ' (' + item.process_code + ')</option>');
                        });
                        processSelect.prop('disabled', false);
                    } else {
                        processSelect.append('<option value="">Tidak ada process untuk kriteria ini</option>');
                    }
                },
                error: function() {
                    alert("Gagal mengambil data proses dari server!");
                }
            });
        } else {
            processSelect.empty().append('<option value="" selected>Pilih device & tipe terlebih dahulu</option>');
        }
    }

    // Trigger saat Device berubah
    $('#device').change(function() {
        if ($(this).val() !== '') {
            $('#type').prop('disabled', false);
            if ($('#type').val() === '') {
                $('#type').val(''); // Reset tipe jika baru pilih device
            }
        } else {
            $('#type').val('').prop('disabled', true);
        }
        loadProcesses();
    });

    // Trigger saat Tipe berubah
    $('#type').change(function() {
        loadProcesses();
    });
});
</script>
